""" 코일 거리별 단위중량 편차(%) 자가 진화형 예측 모델 학습·검증 스크립트 (Google Colab용) 핵심 개선 사항: 1. 질량 보존 법칙(Mass Conservation) 강제 적용: - 코일 실중량(kg) / 총 길이(m)로 계산된 코일 평균 편차(mean_dev)를 기준선으로 고정하고, 모델은 평균 대비 구간별 형상 변화(y - mean_dev)만 예측 후 적분 평균을 mean_dev에 정확히 일치시킵니다. 2. 15개 물리·공정 피처 자동 추출: - 꼬리부/선단부 다중 시정수 과도응답, 4차 다항 기저, 꼬리부 실측 앵커, IBA 온도(BAR_TEMP)·온도구배·토크/전류(TORQUE/CURRENT)·선속(FB_RPM) 시계열 보간 3. 다중 알고리즘 + 후진 피처 소거법(Backward Feature Selection) 자동 진화: - Ridge(다중 alpha), BayesianRidge, HuberRegressor를 Leave-One-Coil-Out 교차검증(LOCO-CV)으로 자동 비교하고, 불필요한 피처를 제거하며 오차(MAE)가 가장 낮은 최적 모델과 잔차 보정곡선을 자동 선정합니다. 4. 홈페이지(coil_verification_platform.html)에서 코드 수정 없이 바로 불러올 수 있는 'profile_model_coeffs.json' 파일을 자동 생성합니다. """ valid_idx = np.where(~np.isnan(length_series))[0] if len(valid_idx) < 5: return None, None, np.nan vals = length_series[valid_idx] breaks = np.where(np.diff(vals) < -100.0)[0] starts = np.concatenate(([0], breaks + 1)) ends = np.concatenate((breaks, [len(vals) - 1])) best_span = -1.0 best_range = (None, None, np.nan) for s, e in zip(starts, ends): seg = vals[s : e + 1] if len(seg) < 5: continue span = float(np.nanmax(seg) - np.nanmin(seg)) if span > best_span: best_span = span best_range = (int(valid_idx[s]), int(valid_idx[e]), float(np.nanmax(seg))) return best_range def find_col_index(headers, keywords, exclude_keywords=None): """헤더 목록에서 키워드가 모두 포함된 첫 번째 컬럼 인덱스를 반환합니다.""" exclude_keywords = exclude_keywords or [] for idx, h in enumerate(headers): hu = str(h).upper() if all(k.upper() in hu for k in keywords) and not any( ex.upper() in hu for ex in exclude_keywords ): return idx return -1 def parse_single_coil_excel(filepath: str): """ 단일 코일 엑셀(.xlsx)에서 코일 메타정보, 단위중량 실측 포인트, IBA 시계열을 추출합니다. D12/D10 사이즈 우선 판별 (SD500N D12 오분류 방지) if "D12" in spec_text: spec = "500N D12" elif "D10" in spec_text: spec = "SD500 D10" else: spec = "500N D12" # --- 2) 시트2: 단위중량실측 --- wt_sheet = next( (s for s in sheets if "단위중량" in s), sheets[1] if len(sheets) >= 2 else None ) if wt_sheet is None: return None df_wt = xl.parse(wt_sheet) dist_col = next((c for c in df_wt.columns if "Distance" in str(c)), None) pct_col = next( (c for c in df_wt.columns if "Kg/M" in str(c) and "%" in str(c)), None ) if dist_col is None or pct_col is None: return None wt_clean = df_wt[[dist_col, pct_col]].apply(pd.to_numeric, errors="coerce").dropna() wt_clean = wt_clean.sort_values(by=dist_col) if len(wt_clean) < 5: return None distances = wt_clean[dist_col].to_numpy(dtype=float) actual_pct = wt_clean[pct_col].to_numpy(dtype=float) total_len = float(np.max(distances)) if total_len <= 100.0: return None frac_actual = np.clip(distances / total_len, 0.0, 1.0) tail_dev = float(actual_pct[0]) # 꼬리부(0m 부근) 기준 편차 # 코일 실중량 기반 평균 편차(질량 보존 기준선) 계산 nom_kg_m = NOMINAL_KG_PER_M.get(spec, 0.888) if not np.isnan(coil_weight_kg): mean_dev = ((coil_weight_kg / total_len) / nom_kg_m - 1.0) * 100.0 has_scale_weight = True else: # NumPy 2.0+ 호환 사다리꼴 적분으로 실측 거리 평균 편차 계산 (np.trapz 제거 대응) if len(frac_actual) > 1: df_seg = np.diff(frac_actual) span_f = float(frac_actual[-1] - frac_actual[0]) if span_f > 1e-6: mean_dev = float( np.sum(0.5 * (actual_pct[:-1] + actual_pct[1:]) * df_seg) / span_f ) else: mean_dev = float(np.mean(actual_pct)) else: mean_dev = float(np.mean(actual_pct)) has_scale_weight = False tail_rel = tail_dev - mean_dev # --- 3) 시트3: IBA 시계열 --- temp_rel_at_pts = np.zeros_like(frac_actual) temp_grad_at_pts = np.zeros_like(frac_actual) torque_rel_at_pts = np.zeros_like(frac_actual) speed_rel_at_pts = np.zeros_like(frac_actual) has_iba = False iba_sheet = sheets[2] if len(sheets) >= 3 else next( (s for s in sheets if "IBA" in s.upper()), None ) if iba_sheet is not None: try: raw_iba = xl.parse(iba_sheet, header=None) if len(raw_iba) > 5 and raw_iba.shape[1] >= 2: c0r0 = str(raw_iba.iloc[0, 0]).strip() c0r1 = str(raw_iba.iloc[1, 0]).strip() if c0r0 == "Time" and c0r1 == "time": header_idx, data_idx = 1, 3 elif any("[" in str(v) for v in raw_iba.iloc[1].values): header_idx, data_idx = 0, 2 else: header_idx, data_idx = 0, 1 raw_headers = [str(h).strip() for h in raw_iba.iloc[header_idx].values] norm_headers = [ h.replace("L1.", ".").replace("L2.", ".") for h in raw_headers ] data_df = raw_iba.iloc[data_idx:].reset_index(drop=True) len_idx = find_col_index(norm_headers, ["COIL_LENGTH"]) temp_idx = find_col_index(norm_headers, ["BAR_TEMP"]) fb_idx = find_col_index(norm_headers, ["FINISHING BLOCK", "SPEED"]) if fb_idx < 0: fb_idx = find_col_index(norm_headers, ["SPEED"]) tq_idx = find_col_index(norm_headers, ["FINISHING", "TORQUE"]) if tq_idx < 0: tq_idx = find_col_index(norm_headers, ["TORQUE"]) if tq_idx < 0: tq_idx = find_col_index(norm_headers, ["CURRENT"]) if len_idx >= 0: len_arr = pd.to_numeric( data_df.iloc[:, len_idx], errors="coerce" ).to_numpy(dtype=float) s_idx, e_idx, ramp_max = extract_ramp_segment(len_arr) if s_idx is not None and ramp_max > 100.0: has_iba = True # ============================================================================= # 3. 자가 진화형 모델 탐색 및 코일 단위 교차검증 (LOCO-CV) # ============================================================================= def run_loco_cv(coils: list, model_fn, feat_mask: np.ndarray): """질량 보존 보정(mean_dev 앵커)이 결합된 상태에서 Leave-One-Coil-Out CV를 수행합니다.""" all_abs_err = [] min_abs_err = [] grid_frac = np.linspace(0.0, 1.0, 41) oof_grid_residuals = [] for i in range(len(coils)): train_X = np.vstack([c["X"][:, feat_mask] for j, c in enumerate(coils) if j != i]) train_y = np.concatenate([c["y_shape"] for j, c in enumerate(coils) if j != i]) test_X = coils[i]["X"][:, feat_mask] test_y_raw = coils[i]["y_raw"] test_mean = coils[i]["mean_dev"] test_frac = coils[i]["frac_actual"] model = model_fn() model.fit(train_X, train_y) raw_shape = model.predict(test_X) #
